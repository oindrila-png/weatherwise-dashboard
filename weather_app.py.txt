def get_api_key() -> Optional[str]:
    """
    Retrieve the OpenWeather API key from the environment.

    Returns
    -------
    Optional[str]
        API key if configured, otherwise None.
    """

    api_key = os.getenv("OPENWEATHER_API_KEY", "").strip()

    if not api_key:
        return None

    return api_key


def ensure_directories() -> None:
    """Create application directories if they do not already exist."""

    DATA_DIR.mkdir(parents=True, exist_ok=True)
    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)


def safe_float(value: object, default: float = 0.0) -> float:
    """Convert a value to float without raising an exception."""

    try:
        return float(value)
    except (TypeError, ValueError):
        return default

class WeatherAPIError(Exception):
    """Raised when a weather API operation fails."""


class OpenWeatherClient:
    """Client for the OpenWeather geocoding and current-weather APIs."""

    def __init__(
        self,
        api_key: str,
        timeout: int = REQUEST_TIMEOUT_SECONDS,
    ) -> None:
        self.api_key = api_key
        self.timeout = timeout

        self.session = requests.Session()
        self.session.headers.update(
            {
                "User-Agent": "WeatherMLApp/1.0",
                "Accept": "application/json",
            }
        )

    def _request(
        self,
        url: str,
        params: dict,
    ) -> dict:
        """Execute an HTTP GET request and return JSON."""

        try:
            response: Response = self.session.get(
                url,
                params=params,
                timeout=self.timeout,
            )

        except Timeout as exc:
            raise WeatherAPIError(
                "The weather service timed out. "
                "Please check your internet connection and try again."
            ) from exc

        except RequestException as exc:
            raise WeatherAPIError(
                f"Network error while contacting weather service: {exc}"
            ) from exc

        if response.status_code == 401:
            raise WeatherAPIError(
                "The OpenWeather API key was rejected. "
                "Check OPENWEATHER_API_KEY."
            )

        if response.status_code == 404:
            raise WeatherAPIError(
                "The requested weather resource was not found."
            )

        if response.status_code == 429:
            raise WeatherAPIError(
                "OpenWeather API rate limit reached. "
                "Please wait and try again."
            )

        if not response.ok:
            try:
                error_body = response.json()
            except ValueError:
                error_body = response.text

            raise WeatherAPIError(
                f"OpenWeather returned HTTP {response.status_code}: "
                f"{error_body}"
            )

        try:
            return response.json()
        except ValueError as exc:
            raise WeatherAPIError(
                "OpenWeather returned an invalid JSON response."
            ) from exc

    def geocode_city(self, city: str) -> Location:
        """
        Convert a city name to coordinates using OpenWeather geocoding.
        """

        city = city.strip()

        if not city:
            raise ValueError("City name cannot be empty.")

        params = {
            "q": city,
            "limit": 1,
            "appid": self.api_key,
        }

        data = self._request(
            OPENWEATHER_GEOCODING_URL,
            params,
        )

        if not data:
            raise WeatherAPIError(
                f"No location was found for '{city}'."
            )

        location_data = data[0]

        return Location(
            name=str(location_data.get("name", city)),
            country=str(location_data.get("country", "")),
            state=str(location_data.get("state", "")),
            latitude=safe_float(location_data.get("lat")),
            longitude=safe_float(location_data.get("lon")),
        )

    def get_current_weather(self, city: str) -> WeatherReport:
        """Retrieve current weather for a city."""

        location = self.geocode_city(city)

        params = {
            "lat": location.latitude,
            "lon": location.longitude,
            "appid": self.api_key,
            "units": "metric",
        }

        data = self._request(
            OPENWEATHER_WEATHER_URL,
            params,
        )

        weather_items = data.get("weather", [])

        if not weather_items:
            condition = "Unknown"
            description = "No description available"
        else:
            condition = str(
                weather_items[0].get("main", "Unknown")
            )
            description = str(
                weather_items[0].get(
                    "description",
                    "No description available",
                )
            )

        rainfall = safe_float(
            data.get("rain", {}).get("1h", 0.0)
        )

        timestamp = str(data.get("dt", ""))

        return WeatherReport(
            city=location.name,
            country=location.country,
            temperature=safe_float(
                data.get("main", {}).get("temp")
            ),
            feels_like=safe_float(
                data.get("main", {}).get("feels_like")
            ),
            humidity=int(
                safe_float(
                    data.get("main", {}).get("humidity")
                )
            ),
            wind_speed=safe_float(
                data.get("wind", {}).get("speed")
            ),
            condition=condition,
            description=description,
            rainfall=rainfall,
            timestamp=timestamp,
        )

class WeatherDataset:
    """Loads, validates, cleans, and generates historical weather data."""

    def __init__(self, path: Path = DEFAULT_DATASET_PATH) -> None:
        self.path = path

    def load(self) -> pd.DataFrame:
        """
        Load historical weather data.

        If the CSV does not exist, a synthetic dataset is generated
        and saved for future runs.
        """

        if self.path.exists():
            logger.info("Loading historical dataset: %s", self.path)

            try:
                dataframe = pd.read_csv(self.path)
            except (OSError, pd.errors.ParserError) as exc:
                logger.warning(
                    "Could not read dataset: %s",
                    exc,
                )
                logger.info(
                    "Falling back to synthetic weather data."
                )
                dataframe = self.generate_synthetic_data()
        else:
            logger.info(
                "Historical dataset not found. "
                "Generating synthetic data."
            )

            dataframe = self.generate_synthetic_data()

            try:
                self.path.parent.mkdir(
                    parents=True,
                    exist_ok=True,
                )
                dataframe.to_csv(self.path, index=False)

                logger.info(
                    "Synthetic dataset saved to %s",
                    self.path,
                )

            except OSError as exc:
                logger.warning(
                    "Could not save generated dataset: %s",
                    exc,
                )

        return self.validate_and_clean(dataframe)

    def validate_and_clean(
        self,
        dataframe: pd.DataFrame,
    ) -> pd.DataFrame:
        """Validate required columns and clean invalid rows."""

        missing = REQUIRED_DATA_COLUMNS - set(
            dataframe.columns
        )

        if missing:
            raise ValueError(
                "Dataset is missing required columns: "
                + ", ".join(sorted(missing))
            )

        cleaned = dataframe.copy()

        cleaned["date"] = pd.to_datetime(
            cleaned["date"],
            errors="coerce",
        )

        numeric_columns = [
            "temperature",
            "humidity",
            "wind_speed",
            "rainfall",
        ]

        for column in numeric_columns:
            cleaned[column] = pd.to_numeric(
                cleaned[column],
                errors="coerce",
            )

        cleaned = cleaned.dropna(
            subset=[
                "date",
                *numeric_columns,
            ]
        )

        # Basic physical sanity checks.
        cleaned = cleaned[
            cleaned["humidity"].between(0, 100)
            & (cleaned["wind_speed"] >= 0)
            & (cleaned["rainfall"] >= 0)
        ]

        cleaned = cleaned.sort_values("date")
        cleaned = cleaned.drop_duplicates(
            subset=["date"],
            keep="last",
        )

        if len(cleaned) < 30:
            raise ValueError(
                "At least 30 valid historical observations "
                "are required for ML training."
            )

        return cleaned.reset_index(drop=True)

    @staticmethod
    def generate_synthetic_data(
        days: int = 1095,
        seed: int = RANDOM_STATE,
    ) -> pd.DataFrame:
        """
        Generate realistic-looking synthetic daily weather data.

        This data is intended for demonstration/development when
        no real historical dataset is available. It is NOT a source
        of actual meteorological observations.
        """

        rng = np.random.default_rng(seed)

        dates = pd.date_range(
            end=pd.Timestamp.today().normalize(),
            periods=days,
            freq="D",
        )

        day_of_year = dates.dayofyear.to_numpy()

        # Seasonal temperature component.
        seasonal_temperature = (
            27
            + 6
            * np.sin(
                2 * np.pi * (day_of_year - 80) / 365.25
            )
        )

        # Autocorrelated-ish noise.
        temperature_noise = rng.normal(
            0,
            1.8,
            size=days,
        )

        temperatures = (
            seasonal_temperature
            + temperature_noise
        )

        humidity = np.clip(
            70
            - 0.8 * (
                temperatures - temperatures.mean()
            )
            + rng.normal(0, 7, size=days),
            25,
            98,
        )

        wind_speed = np.clip(
            rng.gamma(
                shape=2.0,
                scale=1.8,
                size=days,
            ),
            0,
            25,
        )

        # More rain during higher-humidity conditions.
        rain_probability = np.clip(
            0.05
            + (humidity - 55) / 250,
            0.05,
            0.45,
        )

        rainy = (
            rng.random(days)
            < rain_probability
        )

        rainfall = np.where(
            rainy,
            rng.gamma(
                shape=1.8,
                scale=4.0,
                size=days,
            ),
            0,
        )

        return pd.DataFrame(
            {
                "date": dates,
                "temperature": temperatures.round(2),
                "humidity": humidity.round(2),
                "wind_speed": wind_speed.round(2),
                "rainfall": rainfall.round(2),
            }
        )

class WeatherPredictor:
    """Random Forest temperature prediction model."""

    def __init__(
        self,
        random_state: int = RANDOM_STATE,
    ) -> None:
        self.random_state = random_state

        self.model = RandomForestRegressor(
            n_estimators=300,
            max_depth=12,
            min_samples_leaf=2,
            random_state=random_state,
            n_jobs=-1,
        )

    @staticmethod
    def prepare_features(
        dataframe: pd.DataFrame,
    ) -> pd.DataFrame:
        """Create model features from weather observations."""

        result = dataframe.copy()

        result["day_of_year"] = (
            result["date"].dt.dayofyear
        )

        result["month"] = (
            result["date"].dt.month
        )

        return result

    def train(
        self,
        dataframe: pd.DataFrame,
        test_size: float = 0.2,
    ) -> ModelResult:
        """Train and evaluate the Random Forest model."""

        prepared = self.prepare_features(dataframe)

        X = prepared[FEATURE_COLUMNS]
        y = prepared[TARGET_COLUMN]

        if len(prepared) < 30:
            raise ValueError(
                "Not enough observations to train the model."
            )

        X_train, X_test, y_train, y_test = train_test_split(
            X,
            y,
            test_size=test_size,
            random_state=self.random_state,
            shuffle=False,
        )

        self.model.fit(X_train, y_train)

        predictions = self.model.predict(X_test)

        mae = mean_absolute_error(
            y_test,
            predictions,
        )

        rmse = float(
            np.sqrt(
                mean_squared_error(
                    y_test,
                    predictions,
                )
            )
        )

        r2 = r2_score(
            y_test,
            predictions,
        )

        evaluation = prepared.iloc[
            len(X_train):
        ].copy()

        evaluation["actual_temperature"] = y_test.to_numpy()
        evaluation["predicted_temperature"] = predictions

        return ModelResult(
            model=self.model,
            test_predictions=predictions,
            actual_values=y_test.to_numpy(),
            mae=mae,
            rmse=rmse,
            r2=r2,
            dataframe=evaluation,
        )

    def predict_future(
        self,
        dataframe: pd.DataFrame,
        days: int,
    ) -> pd.DataFrame:
        """
        Produce simple future temperature estimates.

        Future humidity/wind/rainfall are estimated from historical
        averages. This is intentionally transparent: a real production
        forecasting system should use genuine future meteorological
        features or a dedicated time-series/weather-forecast model.
        """

        if days <= 0:
            raise ValueError(
                "Prediction horizon must be greater than zero."
            )

        latest_date = dataframe["date"].max()

        historical_means = {
            "humidity": dataframe["humidity"].tail(30).mean(),
            "wind_speed": dataframe["wind_speed"].tail(30).mean(),
            "rainfall": dataframe["rainfall"].tail(30).mean(),
        }

        future_dates = pd.date_range(
            start=latest_date + pd.Timedelta(days=1),
            periods=days,
            freq="D",
        )

        future = pd.DataFrame(
            {
                "date": future_dates,
                "humidity": historical_means["humidity"],
                "wind_speed": historical_means["wind_speed"],
                "rainfall": historical_means["rainfall"],
            }
        )

        future["day_of_year"] = (
            future["date"].dt.dayofyear
        )

        future["month"] = (
            future["date"].dt.month
        )

        features = future[FEATURE_COLUMNS]

        future["predicted_temperature"] = (
            self.model.predict(features)
        )

        return future

class WeatherVisualizer:
    """Matplotlib-based visualization utilities."""

    @staticmethod
    def plot_model_predictions(
        result: ModelResult,
        output_path: Path,
    ) -> None:
        """Plot actual versus predicted test temperatures."""

        dataframe = result.dataframe

        plt.figure(figsize=(12, 6))

        plt.plot(
            dataframe["date"],
            dataframe["actual_temperature"],
            label="Actual Temperature",
            color="royalblue",
            linewidth=2,
        )

        plt.plot(
            dataframe["date"],
            dataframe["predicted_temperature"],
            label="Model Prediction",
            color="darkorange",
            linewidth=2,
            linestyle="--",
        )

        plt.title(
            "Historical Temperature vs ML Predictions"
        )
        plt.xlabel("Date")
        plt.ylabel("Temperature (°C)")
        plt.grid(
            True,
            alpha=0.25,
        )
        plt.legend()
        plt.tight_layout()

        try:
            plt.savefig(
                output_path,
                dpi=150,
                bbox_inches="tight",
            )
            logger.info(
                "Prediction plot saved to %s",
                output_path,
            )
        finally:
            plt.show()
            plt.close()

    @staticmethod
    def plot_historical_temperature(
        dataframe: pd.DataFrame,
        output_path: Path,
    ) -> None:
        """Plot historical temperature data."""

        plt.figure(figsize=(12, 6))

        plt.plot(
            dataframe["date"],
            dataframe["temperature"],
            color="seagreen",
            linewidth=1.5,
        )

        plt.title("Historical Temperature")
        plt.xlabel("Date")
        plt.ylabel("Temperature (°C)")
        plt.grid(
            True,
            alpha=0.25,
        )
        plt.tight_layout()

        try:
            plt.savefig(
                output_path,
                dpi=150,
                bbox_inches="tight",
            )
            logger.info(
                "Historical plot saved to %s",
                output_path,
            )
        finally:
            plt.show()
            plt.close()

def print_weather_report(
    report: WeatherReport,
) -> None:
    """Display a formatted weather report."""

    print("\n" + "=" * 60)
    print("LIVE WEATHER REPORT")
    print("=" * 60)

    print(
        f"Location       : {report.city}, {report.country}"
    )
    print(
        f"Temperature    : {report.temperature:.1f} °C"
    )
    print(
        f"Feels Like     : {report.feels_like:.1f} °C"
    )
    print(
        f"Humidity       : {report.humidity}%"
    )
    print(
        f"Wind Speed     : {report.wind_speed:.1f} m/s"
    )
    print(
        f"Condition      : {report.condition}"
    )
    print(
        f"Description    : {report.description.title()}"
    )
    print(
        f"Rainfall (1h)  : {report.rainfall:.1f} mm"
    )
    print("=" * 60)


def print_model_metrics(
    result: ModelResult,
) -> None:
    """Display model evaluation metrics."""

    print("\n" + "=" * 60)
    print("MODEL EVALUATION")
    print("=" * 60)

    print(
        f"MAE  : {result.mae:.3f} °C"
    )
    print(
        f"RMSE : {result.rmse:.3f} °C"
    )
    print(
        f"R²   : {result.r2:.3f}"
    )

    print("=" * 60)


def print_future_predictions(
    predictions: pd.DataFrame,
) -> None:
    """Display future predictions."""

    print("\n" + "=" * 60)
    print("FUTURE TEMPERATURE ESTIMATES")
    print("=" * 60)

    display = predictions[
        [
            "date",
            "predicted_temperature",
        ]
    ].copy()

    display["date"] = display["date"].dt.strftime(
        "%Y-%m-%d"
    )

    display["predicted_temperature"] = (
        display["predicted_temperature"]
        .round(2)
    )

    print(display.to_string(index=False))

    print("=" * 60)

    print(
        "\nNote: future weather features are estimated from "
        "recent historical averages. These ML estimates are "
        "demonstration forecasts, not official meteorological "
        "forecasts."
    )

def live_weather_menu(
    client: Optional[OpenWeatherClient],
) -> None:
    """Handle live-weather lookup."""

    if client is None:
        print(
            "\nOpenWeather API is not configured."
        )
        print(
            "Set the OPENWEATHER_API_KEY environment variable "
            "before using live weather."
        )
        return

    city = input(
        "\nEnter city name: "
    ).strip()

    if not city:
        print("City name cannot be empty.")
        return

    try:
        report = client.get_current_weather(city)
        print_weather_report(report)

    except ValueError as exc:
        print(f"\nInput error: {exc}")

    except WeatherAPIError as exc:
        print(f"\nWeather API error: {exc}")

    except Exception:
        logger.exception(
            "Unexpected error during live weather lookup."
        )
        print(
            "\nAn unexpected error occurred. "
            "Please check the logs and try again."
        )


def machine_learning_menu(
    dataset: WeatherDataset,
) -> None:
    """Handle model training, evaluation, plotting, and prediction."""

    print("\nLoading historical weather data...")

    try:
        dataframe = dataset.load()

    except (ValueError, OSError) as exc:
        print(
            f"\nDataset error: {exc}"
        )
        return

    print(
        f"Loaded {len(dataframe):,} historical observations."
    )

    predictor = WeatherPredictor()

    try:
        result = predictor.train(dataframe)

    except ValueError as exc:
        print(
            f"\nModel training error: {exc}"
        )
        return

    print_model_metrics(result)

    print("\nGenerating prediction visualization...")

    prediction_plot = (
        OUTPUT_DIR / "historical_vs_predictions.png"
    )

    try:
        WeatherVisualizer.plot_model_predictions(
            result,
            prediction_plot,
        )
    except Exception as exc:
        logger.exception(
            "Could not generate prediction plot."
        )
        print(
            f"\nVisualization error: {exc}"
        )

    while True:
        print(
            "\nML OPTIONS"
        )
        print(
            "1. Predict future temperatures"
        )
        print(
            "2. Plot historical temperature"
        )
        print(
            "3. Return to main menu"
        )

        choice = input(
            "\nChoose an option: "
        ).strip()

        if choice == "1":
            horizon_text = input(
                "How many future days? [1-30]: "
            ).strip()

            try:
                horizon = int(horizon_text)

                if not 1 <= horizon <= 30:
                    raise ValueError

                predictions = predictor.predict_future(
                    dataframe,
                    horizon,
                )

                print_future_predictions(
                    predictions
                )

            except ValueError:
                print(
                    "Please enter a whole number from 1 to 30."
                )

        elif choice == "2":
            historical_plot = (
                OUTPUT_DIR / "historical_temperature.png"
            )

            try:
                WeatherVisualizer.plot_historical_temperature(
                    dataframe,
                    historical_plot,
                )
            except Exception as exc:
                logger.exception(
                    "Could not generate historical plot."
                )
                print(
                    f"\nVisualization error: {exc}"
                )

        elif choice == "3":
            break

        else:
            print(
                "Invalid choice. Please select 1, 2, or 3."
            )

def main() -> int:
    """Application entry point."""

    configure_logging()
    ensure_directories()
    print_banner()

    api_key = get_api_key()
